{{- else if contains "NodePort" .Values.service.type }}

Get the application URL:
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "corveil.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT/ui
{{- else if contains "ClusterIP" .Values.service.type }}

Open the UI:
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "corveil.fullname" . }} 8000:{{ .Values.service.port }}
  then browse to http://localhost:8000/ui{{ if .Values.corveil.devAuth }} and choose "Dev login"{{ end }}
{{- end }}

Next steps (docs/GETTING_STARTED.md):
  1. Create an API key in the UI (API Keys). It starts with {{ .Values.corveil.apiKeyPrefix }}-.
  2. Check the install:   CORVEIL_URL=http://localhost:8000 CORVEIL_API_KEY=<key> corveil health
  3. Connect Claude Code: eval "$(corveil env claude-code --mode direct)" && claude
{{- if not .Values.knowledge.sensors.enabled }}
  4. Enable sensors:      --set knowledge.sensors.enabled=true --set knowledge.ontology.sensorBridge=true
{{- end }}
